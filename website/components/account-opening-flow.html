import { AccountOpeningFlow } from '@sorleed/react';
import '@sorleed/react/styles/tokens.css';

export function Example() {
  return (
    <AccountOpeningFlow
      accountType="individual"
      initialCurrency="USD"
      onComplete={(res) => console.log(res.accountNumber, res.status)}
    />
  );
}
