React Usage

App.tsx
import { StatementExportFlow } from '@sorleed/react';
import '@sorleed/react/styles/tokens.css';

export function Example() {
  return (
    <StatementExportFlow
      accountType="business"
      defaultEmail="finance@company.com"
      onExport={(exp) => console.log(exp.fileName, exp.delivery)}
    />
  );
}
