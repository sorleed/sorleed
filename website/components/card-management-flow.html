CardManagementFlow

Card lifecycle control flow composing Card, Tabs, WalletCard, Tooltip, TransferLimitMeter, Input, Select, Alert, Badge, and Button with frosted visual freeze feedback and icon tooltips.

import { CardManagementFlow } from '@sorleed/react';
import '@sorleed/react/styles/tokens.css';

export function Example() {
  return (
    <CardManagementFlow
      accountType="business"
      dailyLimit={5000}
      usedToday={1120}
      onCardAction={({ action, limit }) => console.log(action, limit)}
    />
  );
}
