TransferLimitMeter

Real-time daily and monthly transfer, wire, and ATM cash velocity limit tracker with threshold alerts and reset windows.

Interactive Playground

$42,500 / $50,000

Installation

Terminal
npx @sorleed/cli add transfer-limit-meter

React Usage

App.tsx
import { TransferLimitMeter } from '@sorleed/react';
import '@sorleed/react/styles/tokens.css';

export function Example() {
  return (
    <TransferLimitMeter
      limits={[
        { id: "wire", label: "Daily Wire & ACH", rail: "Fedwire", used: 42500, max: 50000, currency: "USD", resetText: "Resets in 6h 14m" },
        { id: "atm", label: "ATM Cash Withdrawal", rail: "Debit", used: 400, max: 1500, currency: "USD", resetText: "Resets daily" }
      ]}
      onRequestIncrease={(id) => console.log('Increase requested:', id)}
    />
  );
}
