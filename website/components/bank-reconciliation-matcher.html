BankReconciliationMatcher

Two-column bank statement feed vs general ledger reconciliation matcher with discrepancy detection, match/unmatch actions, and status counters.

Installation

Terminal
npx @sorleed/cli add bank-reconciliation-matcher

React Usage

App.tsx
import { BankReconciliationMatcher } from '@sorleed/react';
import '@sorleed/react/styles/tokens.css';
import '@sorleed/react/styles/bank-reconciliation-matcher.css';

export function Example() {
  return (
    <BankReconciliationMatcher
      bankEntries={[{ id: 'b1', date: '2026-11-18', description: 'Stripe Payout #104', amount: 4250 }]}
      ledgerEntries={[{ id: 'l1', date: '2026-11-18', description: 'AR Settlement - Stripe', amount: 4250 }]}
      onMatch={(bankId, ledgerId) => console.log(bankId, ledgerId)}
    />
  );
}
